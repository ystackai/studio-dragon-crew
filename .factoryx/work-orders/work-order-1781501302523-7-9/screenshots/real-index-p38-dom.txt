    const d = nb.getChannelData(0);
    for (let i=0;i<d.length;i++) d[i] = (Math.random()*2-1) * (1 - i/d.length);
    n.buffer = nb;
    const ng = audioCtx.createGain();
    ng.gain.value = 0.22;
    ng.gain.linearRampToValueAtTime(0.0001, now + 0.5);
    n.connect(ng); ng.connect(masterGain);
    n.start(now);
  }

  function updateAudio(dt) {
    if (!audioCtx || !windGain) return;
    const now = audioCtx.currentTime;
    const targetWind = mode === 'playing' ? (0.018 + (speed - BASE_SPEED) * 0.022) : 0.003;
    if (Number.isFinite(targetWind) && Number.isFinite(now)) {
      windGain.gain.setTargetAtTime(Math.max(0.0001, targetWind), now, 0.12);
    }
    if (windFilter) {
      const targetF = 280 + (speed - BASE_SPEED) * 260 + (boost * 180);
      if (Number.isFinite(targetF) && Number.isFinite(now)) {
        windFilter.frequency.setTargetAtTime(Math.max(80, Math.min(18000, targetF)), now, 0.09);
      }
    }
  }

  function setSteerFromClient(clientY) {
    const rect = canvas.getBoundingClientRect();
    const ny = (clientY - rect.top) / rect.height;
    const band = PLAY_BOT - PLAY_TOP;
    steerTarget = Math.max(0, Math.min(1, (ny - PLAY_TOP) / band));
  }

  function requestDash() {
    if (mode !== 'playing') return;
    if (boost < DASH_COST * 0.6) return;
    dashRequested = true;
  }

  function onPointerDown(e) {
    e.preventDefault();
    if (!firstInteraction) {
      firstInteraction = true;
      initAudio();
      rampMaster(0.72, 0.06);
      if (mode === 'ready') startRun();
    }
    pointerDown = true;
    pointerY = e.clientY;
    setSteerFromClient(e.clientY);
    // Quick tap (no move) triggers dash
    setTimeout(() => {
      if (pointerDown && Math.abs(e.clientY - pointerY) < 12) requestDash();
    }, 110);
  }
  function onPointerMove(e) {
    if (!pointerDown) return;
    e.preventDefault();
    setSteerFromClient(e.clientY);
  }
  function onPointerUp(e) {
    e.preventDefault();
    pointerDown = false;
  }

  function onKeyDown(e) {
    keys[e.key] = true;
    if (!firstInteraction) {
      firstInteraction = true;
      initAudio();
      rampMaster(0.72, 0.06);
      if (mode === 'ready') startRun();
    }
    if (mode === 'playing') {
      if (e.key === ' ' || e.key === 'Spacebar') { e.preventDefault(); requestDash(); }
      if (e.key.toLowerCase() === 'r') { /* allow restart later */ }
    }
    if ((mode === 'crashed' || mode === 'ready') && (e.key === ' ' || e.key === 'r' || e.key === 'R' || e.key === 'Enter')) {
      e.preventDefault();
      restart();
    }
  }
  function onKeyUp(e) { keys[e.key] = false; }

  function bindInput() {
    canvas.addEventListener('pointerdown', onPointerDown, { passive: false });
    window.addEventListener('pointermove', onPointerMove, { passive: false });
    window.addEventListener('pointerup', onPointerUp, { passive: false });
    canvas.addEventListener('touchstart', (e) => { onPointerDown({preventDefault:()=>{}, clientY: e.touches[0].clientY}); }, {passive:false});
    window.addEventListener('touchmove', (e) => { onPointerMove({preventDefault:()=>{}, clientY: e.touches[0].clientY}); }, {passive:false});
    window.addEventListener('touchend', onPointerUp, {passive:false});
    window.addEventListener('keydown', onKeyDown);
    window.addEventListener('keyup', onKeyUp);
    // Also allow dash on canvas click when playing
    canvas.addEventListener('click', (e) => {
      if (mode === 'playing' && !pointerDown) requestDash();
      if (mode === 'crashed') restart();
    });
  }

  function updateInput(dt) {
    // Keyboard steer
    let k = 0;
    if (keys['ArrowUp'] || keys['w'] || keys['W']) k -= 1;
    if (keys['ArrowDown'] || keys['s'] || keys['S']) k += 1;
    if (k !== 0) {
      steerTarget = Math.max(0, Math.min(1, steerTarget + k * 1.6 * dt));
    }
    // Pointer already writes steerTarget directly
    // Dash from held space
    if ((keys[' '] || keys['Spacebar']) && mode === 'playing') requestDash();
  }

  function crash() {
    if (mode !== 'playing') return;
    mode = 'crashed';
    // Carry-the-fire bank on crash: the escalation payoff is legible even in defeat (warmth delivered)
    const bn = document.getElementById('carry-bank');
    if (bn) bn.style.display = 'none';
    let banked = 0;
    if (carryFireUntil > time) {
      banked = 2 + (Math.random() < 0.5 ? 1 : 0);
      embers += banked;
    }
    playCrash();
    rampMaster(0.18, 0.4);
    shake = 22;
    flash = 0.9;
    // Big crash burst (gold-tinged if carried)
    for (let i = 0; i < 38; i++) {
      const ang = (i/38) * Math.PI * 2 + (Math.random()-0.5)*0.6;
      const sp = 0.6 + Math.random()*2.8;
      const col = (banked && i % 4 === 0) ? PAL.gold : (i%3===0 ? PAL.gold : PAL.ember);
      spawnParticle(W*0.28, playerScreenY(), Math.cos(ang)*sp, Math.sin(ang)*sp*0.7 - 0.4,
        0.6 + Math.random()*0.7, 1.6 + Math.random()*2.2, col);
    }
    // Show crash UI + carry note if banked (juicy scoring feedback from the clear beat)
    document.getElementById('final-embers').textContent = embers;
    document.getElementById('final-combo').textContent = '×' + combo;
    document.getElementById('final-dist').textContent = Math.floor(distance);
    if (bn && banked > 0) {
      bn.style.display = 'block';
      document.getElementById('banked').textContent = '+' + banked;
    }
    // Session bests (replay juice)
    if (embers > bestEmbers) bestEmbers = embers;
    if (combo > bestCombo) bestCombo = combo;
    document.getElementById('best-embers').textContent = bestEmbers;
    document.getElementById('best-combo').textContent = bestCombo;
    document.getElementById('crash').classList.add('visible');
  }

  }

  function playerScreenY() {
    const bandH = H * (PLAY_BOT - PLAY_TOP);
    return H * PLAY_TOP + playerY * bandH;
  }

    const y = PLAY_TOP + Math.random() * (PLAY_BOT - PLAY_TOP);
    const kind = Math.random() < 0.62 ? 'spire' : 'vent';
    const h = kind === 'spire' ? (0.09 + Math.random()*0.11) : (0.07 + Math.random()*0.06);
    hazards.push({ x: worldX + W * 0.92 + Math.random()*120, y, kind, h, phase: Math.random()*Math.PI*2 });
    nextHazardAt = worldX + HAZARD_SPAWN_MIN + Math.random()*(HAZARD_SPAWN_MAX - HAZARD_SPAWN_MIN);
  }

    const y = PLAY_TOP + 0.08 + Math.random() * (PLAY_BOT - PLAY_TOP - 0.16);
    embersArr.push({ x: worldX + W * 0.9 + Math.random()*80, y, vy: (Math.random()-0.5)*0.018 });
    nextEmberAt = worldX + EMBER_SPAWN + Math.random()*90;
  }

    const y = PLAY_TOP + 0.12 + Math.random() * (PLAY_BOT - PLAY_TOP - 0.24);

    nextGrazeAt = worldX + GRAZE_SPAWN + Math.random()*220;

  function updateWorld(dt) {
    if (mode !== 'playing') return;

    const isBoost = time < boostUntil;
    speed = isBoost ? BOOST_SPEED : BASE_SPEED;

    // Advance
    const move = speed * 62 * dt;
    worldX += move;
    distance += move * 0.021; // tuned meters feel

    // Player motion (inertia + easing to steerTarget)
    const bandH = (PLAY_BOT - PLAY_TOP);
    const tgtY = steerTarget;
    playerVY = Ease.damp(playerVY, (tgtY - playerY) * 2.8, PLAYER_INERTIA, dt);
    playerY += playerVY * dt * 60;
    playerY = Math.max(0.01, Math.min(0.99, playerY));

    // Boost logic
    if (isBoost) {
      boost = Math.max(0.04, boost - 0.9 * dt);
    } else {
      boost = Math.min(1, boost + BOOST_RECHARGE * dt);
    }
    if (dashRequested && boost >= DASH_COST * 0.55 && time > lastDashAt + 0.22) {
      boostUntil = time + DASH_BOOST_SEC;
      boost = Math.max(0.03, boost - DASH_COST);
      lastDashAt = time;
      playDash();
      // Dash juice: forward surge particles + wing wake
      const py = playerScreenY();
      for (let i=0; i<14; i++) {
        spawnParticle(W*0.26, py + (Math.random()-0.5)*28,
          -1.8 - Math.random()*1.4, (Math.random()-0.5)*1.6,
          0.28 + Math.random()*0.25, 1.4 + Math.random()*1.1, PAL.gold);
      }
      shake = Math.max(shake, 4.5);
      flash = Math.max(flash, 0.22);
    }
    dashRequested = false;

    // Combo decay
    comboTimer += dt;
    if (comboTimer > 0.9) {
      comboTimer = 0;
      if (combo > 1) combo = Math.max(1, combo - COMBO_DECAY_PER_SEC * 0.6);
    }

    // Spawns
    if (worldX > nextHazardAt) spawnHazard();
    if (worldX > nextEmberAt) spawnEmber();
    if (worldX > nextGrazeAt && embers > 1) spawnGraze();

    // Move & collide hazards
    for (let i = hazards.length-1; i>=0; i--) {
      const hz = hazards[i];
      const sx = (hz.x - worldX);
      if (sx < -80) { hazards.splice(i,1); continue; }
      // Simple AABB vs dragon core
      const py = playerScreenY();
      const hx = W*0.26 + (sx / (W*0.9)) * (W*0.5);
      const hy = H*PLAY_TOP + hz.y * H*(PLAY_BOT-PLAY_TOP);
      const hh = hz.h * H * 0.9;
      const hit = (Math.abs(hx - W*0.26) < 30) && (Math.abs(py - hy) < hh*0.48);
      if (hit) {
        // consequence
        crash();
        return;
      }
      // Near-miss heat flash (small)
      if (Math.abs(hx - W*0.26) < 60 && Math.abs(py - hy) < hh*0.9) {
        flash = Math.max(flash, 0.08);
      }
      // Perfect weave micro-reward (juicy dodge feel without spamming; small combo nudge + spark)
      if (!hit && Math.abs(hx - W*0.26) < 44 && Math.abs(py - hy) < hh * 0.72) {
        if (Math.random() < 0.38) {
          flash = Math.max(flash, 0.05);
          weaveFlare = Math.max(weaveFlare, 0.65);
          const wy = (py + hy) * 0.5;
          for (let k=0; k<2; k++) {
            spawnParticle(hx + (Math.random()-0.5)*9, wy + (Math.random()-0.5)*7,
              -0.8 - Math.random()*0.6, (Math.random()-0.5)*0.8 - 0.3,
              0.26 + Math.random()*0.18, 0.9 + Math.random()*0.6, PAL.gold);
          }
          if (combo < 4.5) combo = Math.min(MAX_COMBO, combo + 0.12);
          playWeave();
          if (carryFireUntil > time) {
            // carry-the-fire pays forward on skilled weave (afterglow from Maw clear)
            embers++;
            spawnFloatText('+1', hx + 4, wy - 14, 520);
            for (let k=0; k<3; k++) {
              spawnParticle(hx, wy, (Math.random()-0.5)*1.1, -0.7 + Math.random()*0.4,
                0.32, 0.8, PAL.gold);
            }
          }
        }
      }
    }

    // Embers
    for (let i=embersArr.length-1; i>=0; i--) {
      const e = embersArr[i];
      e.x -= move * 0.6; // slower than world for depth
      e.y += e.vy;
      e.vy *= 0.982;
      const sx = (e.x - worldX);
      if (sx < -40) { embersArr.splice(i,1); continue; }
      const py = playerScreenY();
      const ex = W*0.26 + (sx / (W*0.9)) * (W*0.5);
      const ey = H*PLAY_TOP + e.y * H*(PLAY_BOT-PLAY_TOP);
      if (Math.hypot(ex - W*0.26, ey - py) < 30) {
        // collect
        embersArr.splice(i,1);
        embers++;
        const gained = Math.floor(3 + combo * 0.6);
        // combo
        combo = Math.min(MAX_COMBO, combo + 0.7);
        comboTimer = 0;
        // score pop
        spawnFloatText('+' + gained, ex + 8, ey - 18, 820);
        playEmber((ex / W));
        // recharge boost slightly + particles
        boost = Math.min(1, boost + 0.11);
        for (let k=0; k<7; k++) {
          spawnParticle(ex, ey, (Math.random()-0.5)*1.6, (Math.random()-0.5)*1.2 - 0.6,
            0.4 + Math.random()*0.35, 1.1 + Math.random()*1.3, PAL.gold);
        }
        shake = Math.max(shake, 1.5);
      }
    }

    // Grazes (rescue allies)
    for (let i=grazes.length-1; i>=0; i--) {
      const g = grazes[i];
      g.x -= move * 0.52;
      const sx = (g.x - worldX);
      if (sx < -50) { grazes.splice(i,1); continue; }
      const py = playerScreenY();
      const gx = W*0.26 + (sx / (W*0.9)) * (W*0.5);
      const gy = H*PLAY_TOP + g.y * H*(PLAY_BOT-PLAY_TOP);
      if (Math.hypot(gx - W*0.26, gy - py) < 37) {
        grazes.splice(i,1);
        combo = Math.min(MAX_COMBO, combo + 1.6);
        comboTimer = 0;
        embers += 2;
        spawnFloatText('RESCUED', gx + 6, gy - 24, 1100);
        playGraze();
        boost = Math.min(1, boost + 0.19);
        for (let k=0; k<11; k++) {
          spawnParticle(gx, gy, (Math.random()-0.5)*2.1, (Math.random()-0.5)*1.8 - 0.8,
            0.55 + Math.random()*0.4, 1.0 + Math.random()*1.4, k%2 ? PAL.heat : PAL.gold);
        }
        flash = Math.max(flash, 0.18);
      }
    }

    // Sky Maw escalation (clear boss beat: undulating flame serpent with weave gaps)
    if (!maw && mode === 'playing') {
      const firstGate = (distance > 780 && mawClearedAt < 0);
      const secondGate = (mawClearedAt > 0 && distance > mawClearedAt + 95 && distance < mawClearedAt + 320);
      if (firstGate || secondGate) {
        maw = { x: worldX + W * 1.05, phase: secondGate ? 2.4 : 0, passed: false, second: secondGate };
        // dramatic entrance FX (stronger on second)
        const my = playerScreenY();
        const nparts = secondGate ? 11 : 18;
        for (let k=0; k<nparts; k++) {
          spawnParticle(W*0.78, my + (k-9)*3.5, -0.6 - Math.random()*0.8, (Math.random()-0.5)*1.1,
            0.6 + Math.random()*0.5, 2.2 + Math.random()*1.4, k%2 ? PAL.heat : PAL.ember);
        }
        flash = Math.max(flash, secondGate ? 0.28 : 0.35);
        shake = Math.max(shake, secondGate ? 5 : 7);
      }
    }
    if (maw) {
      const phaseRate = 1.8 + (maw.second ? 0.45 : 0); // second pass feels faster, heavier to thread (escalation)
      maw.phase += dt * phaseRate;
      // Occasional vent/gap telegraph particles (readability for "weave the gaps" without clutter)
      if (Math.random() < 0.22) {
        const vx = (maw.x - worldX) / (W * 0.9) * (W * 0.5) + W * 0.26 + (Math.random()-0.5) * 28;
        const vy = H*PLAY_TOP + (0.5 + Math.sin(maw.phase * 1.3) * 0.26) * H*(PLAY_BOT-PLAY_TOP);
        spawnParticle(vx + (Math.random()-0.5)*11, vy + (Math.random()-0.5)*19,
          -1.4 - Math.random()*0.7, (Math.random()-0.5)*1.1 - 0.2,
          0.32 + Math.random()*0.22, 1.1 + Math.random()*0.7, PAL.heat);
      }
      const mMove = speed * 58 * dt;
      maw.x -= mMove * 0.78; // slightly slower for presence/weight
      const sx = (maw.x - worldX);
      if (sx < -W * 0.6 && !maw.passed) {
        // passed cleanly — escalation beat complete (first or second weave)
        maw.passed = true;
        const isSecond = !!maw.second;
        const bonus = (isSecond ? 46 : 28) + Math.floor(combo * (isSecond ? 2.1 : 1.8));
        embers += isSecond ? 5 : 3;
        combo = Math.min(MAX_COMBO, combo + (isSecond ? 3.0 : 2.2));
        comboTimer = 0;
        mawClearedAt = Math.max(mawClearedAt, distance);
        spawnFloatText((isSecond ? 'MAW CLEARED +' : 'MAW CHAIN +') + bonus, W*0.5 - 40, H*0.42, 1400);
        for (let k=0; k<(isSecond?22:16); k++) {
          spawnParticle(W*0.32, H*0.5 + (k-8)*2, -2.2 + Math.random()*1.4, (Math.random()-0.5)*2.6,
            0.7 + Math.random()*0.4, 1.6, PAL.gold);
        }
        flash = Math.max(flash, isSecond ? 0.38 : 0.28);
        if (isSecond) {
          shake = Math.max(shake, 4);
          heroicFlare = 1.15; // dragon crest acknowledges the feat (cost of power paid, warmth earned)
          playMawClear();
          carryFireUntil = time + 8.0; // the fire is carried forward; visual + scoring linger from the clear escalation beat
        }
      }
      if (sx < -180) { maw = null; } // despawn after pass
      else if (mode === 'playing') {
        // Collision with maw body (weave the gaps) — use same projection as drawMaw
        const py = playerScreenY();
        const mHitX = (maw.x - worldX) / (W * 0.9) * (W * 0.5) + W * 0.26;
        const distToMaw = Math.abs(mHitX - W * 0.26);
        if (distToMaw < 60) {
          const my = H*PLAY_TOP + (0.5 + Math.sin(maw.phase * 1.3) * 0.26) * H*(PLAY_BOT-PLAY_TOP);
          const mThick = 42 + Math.sin(maw.phase*2.1) * 11 + (maw.second ? 7 : 0); // second pass: hungrier, narrower mercy (cost)
          if (Math.abs(py - my) < mThick) {
            // hit the serpent body (no safe gap here)
            crash();
            return;
          }
          // near-miss juice on the body edges (feel the heat)
          if (Math.abs(py - my) < mThick + 22) {
            flash = Math.max(flash, 0.09);
          }
        }
      }
    }

    // Carry-the-fire afterglow (post second Maw clear): small gold motes + scoring juice on weaves (the warmth pays forward)
    if (carryFireUntil > time && mode === 'playing') {
      const py = playerScreenY();
      const px = W * 0.26;
      if (Math.random() < 0.85) {
        spawnParticle(px + (Math.random()-0.5)*22, py + (Math.random()-0.5)*16 - 2,
          -1.1 + Math.random()*0.5, (Math.random()-0.5)*0.9,
          0.48 + Math.random()*0.18, 0.95, PAL.gold);
      }
    }

    // Update particles
    for (let i=particles.length-1; i>=0; i--) {
      const p = particles[i];
      p.x += p.vx * (60*dt);
      p.y += p.vy * (60*dt);
      p.vy += 0.018; // light gravity on embers/debris
      p.life -= dt * (0.9 + (p.col === PAL.gold ? 0.3 : 0));
      p.sz *= 0.995;
      if (p.life <= 0.02) {
        particles.splice(i,1);
        partPool.push(p);
      }
    }

    // Decay FX
    shake = Math.max(0, shake * 0.86 - dt * 3);
    flash = Math.max(0, flash - dt * 1.6);
    weaveFlare = Math.max(0, weaveFlare - dt * 2.1);
    heroicFlare = Math.max(0, heroicFlare - dt * 1.7);
  }

      const f = floatTexts[i];
      f.y += f.vy;
      f.life -= dt;
      if (f.life <= 0) floatTexts.splice(i,1);
    }
  }

  function render(t) {
    DPR = Math.min(window.devicePixelRatio || 1, 2);
    if (canvas.width !== Math.floor(W * DPR) || canvas.height !== Math.floor(H * DPR)) {
      canvas.width = Math.floor(W * DPR);
      canvas.height = Math.floor(H * DPR);
      canvas.style.width = W + 'px';
      canvas.style.height = H + 'px';
    }
    ctx.setTransform(DPR, 0, 0, DPR, 0, 0);

    const dt = lastT ? Math.min(0.05, (t - lastT) / 1000) : 0.016;
    lastT = t;
    time += dt;

    // Sim
    updateInput(dt);
    updateWorld(dt);
    updateFloats(dt);
    updateAudio(dt);

    const boosting = time < boostUntil;

    // Camera shake
    const sx = (Math.random() - 0.5) * shake;
    const sy = (Math.random() - 0.5) * shake * 0.6;

    ctx.save();
    ctx.translate(sx, sy);

    // Background layers
    drawBackground(time);

    // Near foreground rock shards (fast parallax, speed feel)
    ctx.fillStyle = 'rgba(18,10,8,0.75)';
    const fgOff = (worldX * 0.9) % 140;
